```python
"""PDF 논문 요약: PAPER_PROMPT + 강건 JSON 파싱 + bridge(claude CLI) 호출.

liby services/ai/claude.py의 순수 부분을 anthropic 의존 없이 복사.
"""
import json
import re
from dataclasses import dataclass, field
from services import bridge_client
import config

_PAPER_TEXT_LIMIT = 45000


@dataclass
class SummaryResult:
    title: str
    summary: str
    sections: list[dict]
    tags: list[str]
    suggested_topic: str
    insights: list[str] | None = None
    questions_raised: list[str] | None = None
    paragraphs: list[dict] = field(default_factory=list)
    cost_usd: float = 0.0


def _parse_json(raw: str) -> dict:
    raw = raw.strip()
    if raw.startswith("```"):
        raw = re.sub(r'^```(?:json)?\n?', '', raw)
        raw = re.sub(r'\n?```\s*$', '', raw)
    raw = raw.strip()
    try:
        return json.loads(raw)
    except json.JSONDecodeError:
        pass
    decoder = json.JSONDecoder()
    candidates: list[dict] = []
    i = 0
    while True:
        start = raw.find("{", i)
        if start == -1:
            break
        try:
            obj, end = decoder.raw_decode(raw, start)
        except json.JSONDecodeError:
            i = start + 1
            continue
        if isinstance(obj, dict):
            candidates.append(obj)
            i = end
        else:
            i = start + 1
    if not candidates:
        raise ValueError(f"JSON 객체를 찾을 수 없습니다: {raw[:200]}")
    _KNOWN = {"sections", "summary", "paragraphs", "title"}
    candidates.sort(key=lambda d: (len(_KNOWN & d.keys()), len(d)), reverse=True)
    return candidates[0]


def _to_t(val) -> int | None:
    try:
        return int(float(val))
    except (TypeError, ValueError):
        return None


def _build_refs(refs_raw: list) -> list[dict]:
    out = []
    for r in refs_raw or []:
        if not isinstance(r, dict):
            continue
        t = _to_t(r.get("t"))
        if t is None:
            continue
        ref = {"t": t}
        snippet = str(r.get("snippet", "")).strip()
        if snippet:
            ref["snippet"] = snippet
        out.append(ref)
    return out


def _build_sections(data: dict) -> list[dict]:
    out = []
    for sec in data.get("sections", []):
        if not isinstance(sec, dict):
            continue
        heading = str(sec.get("heading", "")).strip()
        if not heading:
            continue
        subs = []
        for sub in sec.get("subsections", []):
            if not isinstance(sub, dict):
                continue
            sub_heading = str(sub.get("heading", "")).strip()
            if not sub_heading:
                continue
            items = []
            for it in sub.get("items", []):
                if not isinstance(it, dict):
                    continue
                text = str(it.get("text", "")).strip()
                if not text:
                    continue
                item_obj = {"text": text, "refs": _build_refs(it.get("refs", []))}
                fig = _to_t(it.get("figure"))
                if fig is not None:
                    item_obj["figure"] = fig
                items.append(item_obj)
            subs.append({"heading": sub_heading, "items": items})
        out.append({"heading": heading, "subsections": subs})
    return out


def _build_paragraphs(data: dict) -> list[dict]:
    out = []
    for p in data.get("paragraphs", []):
        if not isinstance(p, dict):
            continue
        text = str(p.get("text", "")).strip()
        if not text:
            continue
        out.append({"text": text, "refs": _build_refs(p.get("refs", []))})
    return out


PAPER_PROMPT = """당신은 박사과정 학생의 논문 분석을 돕는 연구 조교입니다.
아래 학술 논문을 읽고, 박사과정 학생이 지도교수와의 랩미팅에서 **발표할 수 있는 수준**의
한국어 요약 노트를 작성하세요. 표면적 나열이 아니라 핵심을 짚는 깊이가 있어야 합니다.

기존 주제 목록: {existing_topics}

{figures}

논문 본문:
{text}

규칙:
- 반드시 아래 **5개 대섹션을 이 순서·이 제목**으로 작성합니다(누락 금지):
  1. 연구 목적 — 이 논문이 풀려는 문제, 동기, 기여(contribution).
  2. 주요 실험 내용 — 실험/시뮬레이션 설계, 대상, 변수, 방법론.
  3. 이론적 기반 — 바탕이 되는 이론·모델·수식·가정.
  4. 실험 결과 — 핵심 수치·경향·그래프가 말하는 바, 정량적 결과.
  5. 생각해볼 점 — 한계, 의의, 후속 연구 방향, 랩미팅에서 던질 만한 비판적 질문.
- 각 대섹션은 1~3개의 소섹션(heading: "1.1 ...")을 가지며, 각 소섹션은 2~4개의 문단형 항목(items)을 가집니다.
- 각 item의 text는 2~4문장의 충실한 한국어 설명입니다. 전문 용어는 그대로 쓰되 풀어서 설명합니다.
- 그림 배치: 위 "사용 가능한 그림"의 번호를 참고해, 그림을 언급·설명하는 문단 item에 "figure": 그림번호 를 넣습니다.
  해당 그림이 그 문단 바로 아래에 표시됩니다. 결과/실험 섹션에 그림을 적극 배치하세요. 그림이 없으면 figure 키를 생략합니다.
- refs는 항상 빈 리스트로 둡니다(논문은 타임스탬프가 없음).

JSON으로만 응답하세요:
{{"title": "한국어 논문 제목(원제 의미를 살린)", "language": "ko",
  "summary": "논문 전체를 아우르는 2~3문장 한국어 핵심 요약",
  "tags": ["키워드1", "키워드2", "키워드3"],
  "suggested_topic": "기존_주제_중_하나_또는_새_주제명",
  "sections": [
    {{"heading": "1. 연구 목적", "subsections": [
      {{"heading": "1.1 문제 정의", "items": [
        {{"text": "2~4문장 한국어 설명", "refs": []}},
        {{"text": "그림을 설명하는 문단", "refs": [], "figure": 2}}
      ]}}
    ]}},
    {{"heading": "2. 주요 실험 내용", "subsections": [...]}},
    {{"heading": "3. 이론적 기반", "subsections": [...]}},
    {{"heading": "4. 실험 결과", "subsections": [...]}},
    {{"heading": "5. 생각해볼 점", "subsections": [...]}}
  ]}}"""


async def summarize_paper(
    text: str, figures_manifest: str, existing_topics: list[str] | None = None,
) -> SummaryResult:
    figures_block = (
        f"사용 가능한 그림(번호와 캡션):\n{figures_manifest}\n"
        if figures_manifest else "사용 가능한 그림: 없음\n"
    )
    prompt = PAPER_PROMPT.format(
        text=text[:_PAPER_TEXT_LIMIT],
        existing_topics=", ".join(existing_topics or []) or "없음",
        figures=figures_block,
    )
    last_err = "알 수 없음"
    for _ in range(2):
        run = await bridge_client.run_agent(
            prompt, adapter=config.BRIDGE_ADAPTER, cwd=config.BRIDGE_CWD,
            timeout_sec=config.BRIDGE_TIMEOUT_SEC,
        )
        try:
            data = _parse_json(run.summary)
        except (ValueError, json.JSONDecodeError):
            last_err = f"JSON 파싱 실패: {run.summary[:150]}"
            continue
        sections = _build_sections(data)
        if not sections and not data.get("summary"):
            last_err = f"빈 응답: {run.summary[:150]}"
            continue
        break
    else:
        raise ValueError(f"논문 분석 실패(2회 시도): {last_err}")
    return SummaryResult(
        title=data.get("title", "제목 없음"),
        summary=data.get("summary", ""),
        sections=sections,
        tags=data.get("tags", []),
        suggested_topic=data.get("suggested_topic", ""),
        insights=data.get("insights"),
        questions_raised=data.get("questions_raised"),
        paragraphs=_build_paragraphs(data),
        cost_usd=run.usage.total_cost_usd,
    )
```

```python
import json
import aiosqlite
from services.summarizer import SummaryResult

_JSON_FIELDS = ["sections", "tags", "insights", "questions_raised", "paragraphs"]


def _parse_row(row: dict) -> dict:
    for f in _JSON_FIELDS:
        if isinstance(row.get(f), str):
            try:
                row[f] = json.loads(row[f])
            except (json.JSONDecodeError, TypeError):
                pass
    return row


async def save_note(db_path: str, source_url: str, result: SummaryResult) -> int:
    async with aiosqlite.connect(db_path) as db:
        cur = await db.execute(
            """INSERT INTO items
               (type, title, source_url, summary, sections, tags, topic,
                summary_mode, insights, questions_raised, paragraphs)
               VALUES ('pdf',?,?,?,?,?,?, 'detailed', ?,?,?)""",
            (
                result.title, source_url, result.summary,
                json.dumps(result.sections, ensure_ascii=False),
                json.dumps(result.tags, ensure_ascii=False),
                result.suggested_topic,
                json.dumps(result.insights or [], ensure_ascii=False),
                json.dumps(result.questions_raised or [], ensure_ascii=False),
                json.dumps(result.paragraphs or [], ensure_ascii=False),
            ),
        )
        await db.commit()
        return cur.lastrowid


async def get_note(db_path: str, note_id: int) -> dict | None:
    async with aiosqlite.connect(db_path) as db:
        db.row_factory = aiosqlite.Row
        cur = await db.execute("SELECT * FROM items WHERE id=?", (note_id,))
        row = await cur.fetchone()
    return _parse_row(dict(row)) if row else None


async def list_notes(db_path: str, search: str | None = None, limit: int = 100) -> list[dict]:
    query = "SELECT * FROM items WHERE 1=1"
    params: list = []
    if search:
        pat = f"%{search.lower()}%"
        query += " AND (LOWER(title) LIKE ? OR LOWER(summary) LIKE ? OR LOWER(tags) LIKE ?)"
        params += [pat, pat, pat]
    query += " ORDER BY created_at DESC LIMIT ?"
    params.append(limit)
    async with aiosqlite.connect(db_path) as db:
        db.row_factory = aiosqlite.Row
        cur = await db.execute(query, params)
        rows = await cur.fetchall()
    return [_parse_row(dict(r)) for r in rows]


async def count_notes(db_path: str) -> int:
    async with aiosqlite.connect(db_path) as db:
        cur = await db.execute("SELECT COUNT(*) FROM items")
        return (await cur.fetchone())[0]


async def delete_note(db_path: str, note_id: int) -> None:
    async with aiosqlite.connect(db_path) as db:
        await db.execute("DELETE FROM items WHERE id=?", (note_id,))
        await db.commit()
```

```python
import asyncio
import logging
import uuid
from dataclasses import dataclass
from typing import Awaitable, Callable

log = logging.getLogger(__name__)


@dataclass
class Task:
    id: str
    title: str
    status: str = "queued"   # queued|running|done|error
    progress: str = "대기 중..."
    error: str | None = None


_tasks: dict[str, Task] = {}
_queue: asyncio.Queue = asyncio.Queue()
DoWork = Callable[[Task], Awaitable[None]]


def new_task(title: str) -> Task:
    t = Task(id=uuid.uuid4().hex[:8], title=title)
    _tasks[t.id] = t
    return t


def get_task(task_id: str) -> Task | None:
    return _tasks.get(task_id)


async def enqueue(task: Task, work: DoWork) -> None:
    await _queue.put((task, work))


async def run_worker() -> None:
    while True:
        task, work = await _queue.get()
        task.status = "running"
        task.progress = "AI 분석 중..."
        try:
            await work(task)
            task.status = "done"
        except Exception as e:  # noqa: BLE001
            log.error("task %s failed: %s", task.id, e, exc_info=True)
            task.status = "error"
            task.error = str(e)
        finally:
            _queue.task_done()
```

import os
import re
import tempfile
import uuid
from fastapi import APIRouter, Form, Request, UploadFile, File
import config
from services.extractor import extract_pdf
from services import pdf_figures, storage, summarizer
from services.task_queue import new_task, enqueue, get_task
from templates_env import templates

router = APIRouter(prefix="/api/pdf", tags=["pdf"])


def _figure_slug(filename: str) -> str:
    base = re.sub(r"\.pdf$", "", filename, flags=re.IGNORECASE)
    base = re.sub(r"[^A-Za-z0-9가-힣]+", "-", base).strip("-")[:40] or "paper"
    return f"{base}-{uuid.uuid4().hex[:6]}"


@router.post("")
async def analyze_pdf(request: Request, file: UploadFile = File(...)):
    file_bytes = await file.read()
    filename = file.filename or "unknown.pdf"
    task = new_task(filename)

    async def do_work(t):
        t.progress = "PDF 텍스트·그림 추출 중..."
        with tempfile.NamedTemporaryFile(delete=False, suffix=".pdf") as tmp:
            tmp.write(file_bytes)
            tmp_path = tmp.name
        try:
            text = await extract_pdf(tmp_path)
            fig_root = os.path.join(config.VAULT_PATH, "pdf")
            slug = _figure_slug(filename)
            try:
                figures = pdf_figures.extract_figures(tmp_path, fig_root, slug)
            except Exception:
                figures = []
        finally:
            os.unlink(tmp_path)
        t.progress = "AI 분석 중(논문 5개 항목)..."
        existing = [n["topic"] for n in await storage.list_notes(config.DB_PATH) if n.get("topic")]
        result = await summarizer.summarize_paper(
            text, pdf_figures.figures_manifest(figures), list(set(existing)))
        if result.sections and figures:
            placed = pdf_figures.attach_figures_to_sections(result.sections, figures)
            gallery = pdf_figures.build_gallery_section(figures, placed)
            if gallery:
                result.sections.append(gallery)
        t.title = result.title
        t.progress = "저장 중..."
        await storage.save_note(config.DB_PATH, filename, result)

    await enqueue(task, do_work)
    return templates.TemplateResponse(request, "partials/task_card.html", {"task": task})


@router.get("/tasks/{task_id}")
async def task_status(request: Request, task_id: str):
    from fastapi.responses import HTMLResponse
    task = get_task(task_id)
    if not task:
        return HTMLResponse('<p class="text-sm text-red-400 py-2">작업을 찾을 수 없습니다.</p>')
    if task.status == "done":
        resp = HTMLResponse("")
        resp.headers["HX-Trigger"] = "noteCompleted"
        return resp
    if task.status == "error":
        return HTMLResponse(
            f'<div id="task-{task.id}" class="bg-red-50 border border-red-200 rounded-xl p-4 dark:bg-red-900/20 dark:border-red-800">'
            f'<div class="flex items-start gap-2"><span class="flex-1 text-sm text-red-600 dark:text-red-400">분석 실패: {task.error}</span>'
            f'<button onclick="this.closest(\'[id^=task-]\').remove()" class="flex-shrink-0 text-[13px] text-red-400 hover:text-red-600 px-1" title="닫기">✕</button></div></div>'
        )
    return templates.TemplateResponse(request, "partials/task_card.html", {"task": task})

import os
import shutil
from pathlib import Path
from typing import Optional
from fastapi import APIRouter, Query, Request
from fastapi.responses import HTMLResponse, RedirectResponse
import config
from services.storage import get_note, list_notes, delete_note
from templates_env import templates

router = APIRouter(prefix="/api/items", tags=["items"])


@router.get("")
async def get_items(request: Request, search: Optional[str] = Query(None)):
    notes = await list_notes(config.DB_PATH, search=search)
    return templates.TemplateResponse(request, "partials/note_list.html", {"notes": notes})


@router.get("/{note_id}/detail")
async def get_item_detail(request: Request, note_id: int):
    note = await get_note(config.DB_PATH, note_id)
    return templates.TemplateResponse(request, "partials/note_detail_modal.html", {"note": note})


@router.get("/{note_id}/read")
async def read_view(request: Request, note_id: int):
    note = await get_note(config.DB_PATH, note_id)
    if not note:
        return RedirectResponse("/", status_code=302)
    return templates.TemplateResponse(request, "read.html", {"note": note})


def _figure_dir_of(note: dict | None) -> str | None:
    if not note:
        return None
    for sec in note.get("sections") or []:
        for sub in sec.get("subsections", []):
            for it in sub.get("items", []):
                img = it.get("image")
                if img and img.get("file") and "/" in img["file"]:
                    return os.path.join(config.VAULT_PATH, "pdf", img["file"].split("/", 1)[0])
    return None


@router.delete("/{note_id}")
async def delete_item(note_id: int):
    note = await get_note(config.DB_PATH, note_id)
    fig_dir = _figure_dir_of(note)
    await delete_note(config.DB_PATH, note_id)
    if fig_dir and os.path.isdir(fig_dir):
        shutil.rmtree(fig_dir, ignore_errors=True)
    resp = HTMLResponse("", status_code=200)
    resp.headers["HX-Trigger"] = "noteDeleted"
    return resp
```
